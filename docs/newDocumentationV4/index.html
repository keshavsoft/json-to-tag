03 · VANILLA JAVASCRIPT

Create the same piece with JavaScript

const wrapper = document.createElement("div");
wrapper.id = "test-content";

const label = document.createElement("label");
label.textContent = "Name";

const text = document.createElement("p");
text.textContent = "Test Content";

wrapper.append(label, text);

document.querySelector("#vanilla-output")
  .replaceChildren(wrapper);

02 · CDN BUILD

Build the same piece

<script src="https://keshavsoft.github.io/json-to-tag/dist/v4/min.js"></script>

const spec = {
  tagName: "div",
  attributes: { id: "test-content" },
  children: [
    { tagName: "label", textContent: "Name" },
    { tagName: "p", textContent: "Test Content" }
  ]
};

const element =
  window.ks.jsonToTag.buildSpecElement(spec);

document.querySelector("#build-output")
  .replaceChildren(element);
